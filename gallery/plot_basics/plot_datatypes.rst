    temp_max = [28.1, 29.4, 31.2, 33.6, 34.9, 35.7, 33.2]

    # 统计五件套：sum / len / max / min + 列表推导式筛选高温日
    t_mean = sum(temp_max) / len(temp_max)
    t_high, t_low = max(temp_max), min(temp_max)
    high_days = [t for t in temp_max if t >= 35]   # 高温日判据：≥ 35 °C
